// أضِف هذه الدوال داخل match /databases/{db}/documents ثم أضف  && photoOk()  لشرط تحديث users/{uid} الموجود عندك.
// الهدف: صورة البروفايل تُحفظ مرة واحدة ولا تتغير إلا إذا سمح المالك (photoReset=true) أو المالك نفسه عدّلها.

function isOwner() { return request.auth != null && request.auth.token.email == 'khwailedapp@gmail.com'; }
function isAdmin() { return request.auth != null && (isOwner() || get(/databases/$(db)/documents/users/$(request.auth.uid)).data.role == 'admin'); }
function changed() { return request.resource.data.diff(resource.data).affectedKeys(); }
function photoOk() {
  return isOwner()
    || !changed().hasAny(['photo','photoLocked','photoReset','photoAt'])
    || ( changed().hasOnly(['photo','photoLocked','photoReset','photoAt'])
         && (resource.data.get('photo','') == '' || resource.data.get('photoReset', false) == true)
         && request.resource.data.photoLocked == true
         && request.resource.data.get('photoReset', false) == false );
}

// السماح للطالب بتحديث أوسمته (users/{uid}.badges) فقط بدون المساس بحقول الصلاحيات أو الرصيد
function badgesOk() {
  return isAdmin()
    || !changed().hasAny(['badges'])
    || (request.auth != null && request.auth.uid == uid && request.resource.data.badges is map
        && !changed().hasAny(['role','vip','vipUntil','wallet','gban','codeAdmin','ownerCode']));
}

// تقييمات الدورات: تقييم واحد لكل طالب في كل دورة بمعرف ثابت {uid}_{courseId}
match /courseReviews/{reviewId} {
  allow read: if request.auth != null;
  allow create, update: if request.auth != null
    && reviewId == request.auth.uid + '_' + request.resource.data.courseId
    && request.resource.data.id == reviewId
    && request.resource.data.uid == request.auth.uid
    && request.resource.data.courseId is string
    && request.resource.data.rating is int
    && request.resource.data.rating >= 1
    && request.resource.data.rating <= 5
    && request.resource.data.comment is string
    && request.resource.data.comment.size() >= 3
    && request.resource.data.comment.size() <= 400;
  allow delete: if isAdmin() || (request.auth != null && resource.data.uid == request.auth.uid);
}

// إعدادات التطبيق والتحديثات عن بُعد (settings/app)
match /settings/app {
  allow read: if true;
  allow write: if isAdmin();
}
